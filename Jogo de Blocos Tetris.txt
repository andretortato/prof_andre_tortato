    <script>
        /* ==========================================================================
           SINTETIZADOR DE ÁUDIO VIA WEB AUDIO API (SEM DEPENDÊNCIAS EXTERNAS)

        class SoundFX {
            constructor() {
                this.ctx = null;
                this.enabled = true;
            }

            init() {
                if (!this.ctx) {
                    const AudioCtx = window.AudioContext || window.webkitAudioContext;
                    if (AudioCtx) this.ctx = new AudioCtx();
                }
                if (this.ctx && this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }

            playTone(freq, type, duration, startVol = 0.1, endVol = 0.001) {
                if (!this.enabled) return;
                this.init();
                if (!this.ctx) return;

                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = type;
                    osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
                    gain.gain.setValueAtTime(startVol, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(endVol, this.ctx.currentTime + duration);

                    osc.connect(gain);
                    gain.connect(this.ctx.destination);

                    osc.start();
                    osc.stop(this.ctx.currentTime + duration);
                } catch (e) { console.error(e); }
            }

            move() { this.playTone(300, 'sine', 0.08, 0.08); }
            rotate() { this.playTone(450, 'triangle', 0.1, 0.1); }
            drop() { this.playTone(180, 'square', 0.1, 0.08); }
            hardDrop() {
                this.playTone(120, 'sawtooth', 0.15, 0.2);
                this.playTone(80, 'sine', 0.2, 0.25);
            }
            hold() { this.playTone(520, 'sine', 0.12, 0.1); }
            
            lineClear() {
                if (!this.enabled) return;
                const notes = [400, 500, 600, 800];
                notes.forEach((freq, idx) => {
                    setTimeout(() => this.playTone(freq, 'sine', 0.15, 0.15), idx * 60);
                });
            }

            stageWin() {
                if (!this.enabled) return;
                const notes = [523.25, 659.25, 783.99, 1046.50];
                notes.forEach((freq, idx) => {
                    setTimeout(() => this.playTone(freq, 'triangle', 0.25, 0.2), idx * 120);
                });
            }

            gameOver() {
                if (!this.enabled) return;
                const notes = [400, 350, 300, 220];
                notes.forEach((freq, idx) => {
                    setTimeout(() => this.playTone(freq, 'sawtooth', 0.25, 0.15), idx * 120);
                });
            }
        }

                    dropInterval = Math.max(200, 800 - (stage - 1) * 40);
                    maxTime = 120;
                    targetLines = 5 + (stage - 1) * 3;
                } else if (difficulty === 'medium') {
                    dropInterval = Math.max(150, 550 - (stage - 1) * 50);
                    maxTime = 90;
                    targetLines = 8 + (stage - 1) * 4;
                } else { // Hard
                    dropInterval = Math.max(100, 350 - (stage - 1) * 40);
                    maxTime = 60;
                    targetLines = 10 + (stage - 1) * 5;
                }
                document.getElementById('timerContainer').style.display = 'block';
            } else { // Endless
                dropInterval = Math.max(100, 700 - Math.floor(linesCleared / 10) * 50);
                document.getElementById('timerContainer').style.display = 'none';
            }

            timeRemaining = maxTime;
            updateUI();
        }

            isStageWin = false;

            startTimer();
        }

            clearInterval(timerInterval);
            if (gameMode !== 'stage') return;

            timerInterval = setInterval(() => {
                if (!isPaused && !isGameOver && !isStageWin) {
                    timeRemaining--;
                    updateTimerUI();

                    if (timeRemaining <= 0) {
                        triggerGameOver("O tempo acabou!");
                    }
                }
            }, 1000);
        }

        /* ==========================================================================
           MOTOR DE TETRIS (GRID, PEÇAS, ROTAÇÃO, COLISÃO)
           ========================================================================== */
        function createGrid() {
            return Array.from({ length: ROWS }, () => Array(COLS).fill(0));
        }

        function getRandomPiece() {
            if (bag.length === 0) {
                bag = ['I', 'J', 'L', 'O', 'S', 'T', 'Z'];
                // Shuffle bag (Fisher-Yates)
                for (let i = bag.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [bag[i], bag[j]] = [bag[j], bag[i]];
                }
            }
            const type = bag.pop();
            const shape = SHAPES[type];
            return {
                type: type,
                shape: shape,
                x: Math.floor((COLS - shape[0].length) / 2),
                y: 0
            };
        }

        function collide(board, piece, offsetX = 0, offsetY = 0, customShape = null) {
            const shape = customShape || piece.shape;
            if (!shape) return true;

            for (let r = 0; r < shape.length; r++) {
                for (let c = 0; c < shape[r].length; c++) {
                    if (shape[r][c]) {
                        const newX = piece.x + c + offsetX;
                        const newY = piece.y + r + offsetY;

                        if (newX < 0 || newX >= COLS || newY >= ROWS) return true;
                        if (newY >= 0 && board[newY] && board[newY][newX]) return true;
                    }
                }
            }
            return false;
        }

        function rotate(matrix) {
            const N = matrix.length;
            const result = Array.from({ length: N }, () => Array(N).fill(0));
            for (let r = 0; r < N; r++) {
                for (let c = 0; c < N; c++) {
                    result[c][N - 1 - r] = matrix[r][c];
                }
            }
            return result;
        }

        function playerRotate() {
            if (isPaused || isGameOver || isStageWin) return;
            const rotated = rotate(currentPiece.shape);
            
            // Tentativa simples de Wall Kick
            let offset = 0;
            if (!collide(grid, currentPiece, 0, 0, rotated)) {
                currentPiece.shape = rotated;
                soundFX.rotate();
            } else if (!collide(grid, currentPiece, -1, 0, rotated)) {
                currentPiece.x -= 1;
                currentPiece.shape = rotated;
                soundFX.rotate();
            } else if (!collide(grid, currentPiece, 1, 0, rotated)) {
                currentPiece.x += 1;
                currentPiece.shape = rotated;
                soundFX.rotate();
            }
        }

        function playerMove(dir) {
            if (isPaused || isGameOver || isStageWin) return;
            if (!collide(grid, currentPiece, dir, 0)) {
                currentPiece.x += dir;
                soundFX.move();
            }
        }

        function playerDrop() {
            if (isPaused || isGameOver || isStageWin) return;
            if (!collide(grid, currentPiece, 0, 1)) {
                currentPiece.y++;
                dropCounter = 0;
            } else {
                lockPiece();
            }
        }

        function playerHardDrop() {
            if (isPaused || isGameOver || isStageWin) return;
            while (!collide(grid, currentPiece, 0, 1)) {
                currentPiece.y++;
                score += 2;
            }
            soundFX.hardDrop();
            triggerScreenShake();
            lockPiece();
        }

        function playerHold() {
            if (isPaused || isGameOver || isStageWin || !canHold) return;

            soundFX.hold();
            if (!holdPiece) {
                holdPiece = currentPiece.type;
                currentPiece = nextPiece;
                nextPiece = getRandomPiece();
            } else {
                const temp = holdPiece;
                holdPiece = currentPiece.type;
                currentPiece = {
                    type: temp,
                    shape: SHAPES[temp],
                    x: Math.floor((COLS - SHAPES[temp][0].length) / 2),
                    y: 0
                };
            }
            canHold = false;
            updateUI();
        }

        function lockPiece() {
            // Fixa a peça no grid
            for (let r = 0; r < currentPiece.shape.length; r++) {
                for (let c = 0; c < currentPiece.shape[r].length; c++) {
                    if (currentPiece.shape[r][c]) {
                        const boardY = currentPiece.y + r;
                        const boardX = currentPiece.x + c;
                        if (boardY < 0) {
                            triggerGameOver("A arena lotou!");
                            return;
                        }
                        grid[boardY][boardX] = currentPiece.type;
                    }
                }
            }

            soundFX.drop();
            checkLineClears();

            currentPiece = nextPiece;
            nextPiece = getRandomPiece();
            canHold = true;

            // Se ao nascer já colidir -> Game Over
            if (collide(grid, currentPiece)) {
                triggerGameOver("A arena lotou!");
            }

            updateUI();
        }

        function checkLineClears() {
            let linesClearedThisTurn = 0;
            let clearedRows = [];

            for (let r = ROWS - 1; r >= 0; r--) {
                if (grid[r].every(cell => cell !== 0)) {
                    linesClearedThisTurn++;
                    clearedRows.push(r);
                }
            }

            if (linesClearedThisTurn > 0) {
                soundFX.lineClear();

                // Efeito de partículas
                clearedRows.forEach(r => {
                    for (let c = 0; c < COLS; c++) {
                        createParticles((c + 0.5) * (canvas.width / COLS), (r + 0.5) * (canvas.height / ROWS), PALETTE[grid[r][c]]?.main || '#fff');
                    }
                });

                // Remover linhas
                for (let i = 0; i < linesClearedThisTurn; i++) {
                    grid.splice(clearedRows[i], 1);
                    grid.unshift(Array(COLS).fill(0));
                }

                // Pontuação Tetris
                const linePoints = [0, 100, 300, 500, 800];
                const pts = (linePoints[linesClearedThisTurn] || 100) * stage;
                score += pts;
                linesCleared += linesClearedThisTurn;

                // Texto Flutuante
                showFloatingText(`+${pts}`, canvas.width / 2, (clearedRows[0] || 5) * (canvas.height / ROWS));

                if (score > highScore) {
                    highScore = score;
                    localStorage.setItem('tetris_ludico_highscore', highScore);
                }

                // Checar Vitória da Fase
                if (gameMode === 'stage' && linesCleared >= targetLines) {
                    triggerStageWin();
                }

                // Atualizar velocidade no modo infinito
                if (gameMode === 'endless') {
                    dropInterval = Math.max(100, 700 - Math.floor(linesCleared / 10) * 50);
                }
            }
        }

        /* ==========================================================================
           SISTEMA DE VITÓRIA, DERROTA E TRANSIÇÃO DE FASES

        function triggerStageWin() {
            isStageWin = true;
            clearInterval(timerInterval);
            soundFX.stageWin();

            // Confetes festivos!
            if (window.confetti) {
                confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });
            }

            const timeBonus = timeRemaining * 50;
            score += timeBonus;
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('tetris_ludico_highscore', highScore);
            }

            // Calcular estrelas
            const starsContainer = document.getElementById('starsContainer');
            const pctTimeLeft = timeRemaining / maxTime;
            let starCount = 1;
            if (pctTimeLeft > 0.5) starCount = 3;
            else if (pctTimeLeft > 0.25) starCount = 2;

            starsContainer.innerHTML = '';
            for (let i = 0; i < 3; i++) {
                if (i < starCount) {
                    starsContainer.innerHTML += '<i class="fa-solid fa-star text-amber-400 drop-shadow"></i>';
                } else {
                    starsContainer.innerHTML += '<i class="fa-solid fa-star text-slate-700"></i>';
                }
            }

            document.getElementById('lblTimeBonus').innerText = `+${timeBonus}`;
            document.getElementById('lblWinScore').innerText = score;

            openModal('winModal');
        }

        /* ==========================================================================
           RENDERIZAÇÃO GRÁFICA (BLOCOS 3D, SOMBRA GUIA, PARTÍCULAS)
           ========================================================================== */
        function gameLoop(time = 0) {
            const deltaTime = time - lastTime;
            lastTime = time;

            if (!isPaused && !isGameOver && !isStageWin && currentPiece) {
                dropCounter += deltaTime;
                if (dropCounter > dropInterval) {
                    playerDrop();
                }
            }

            render();
            updateParticles();
            requestAnimationFrame(gameLoop);
        }

        function render() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const blockWidth = canvas.width / COLS;
            const blockHeight = canvas.height / ROWS;

            // Renderizar Grade Suave de Fundo
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
            ctx.lineWidth = 1;
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    ctx.strokeRect(c * blockWidth, r * blockHeight, blockWidth, blockHeight);
                }
            }

            // Renderizar Blocos Fixos do Grid
            for (let r = 0; r < ROWS; r++) {
                if (!grid[r]) continue;
                for (let c = 0; c < COLS; c++) {
                    if (grid[r][c]) {
                        drawBlock(ctx, c * blockWidth, r * blockHeight, blockWidth, blockHeight, grid[r][c]);
                    }
                }
            }

            if (currentPiece && !isGameOver && !isStageWin) {
                // Renderizar Peça Fantasma (Ghost Piece)
                let ghostY = currentPiece.y;
                while (!collide(grid, currentPiece, 0, ghostY - currentPiece.y + 1)) {
                    ghostY++;
                }

                for (let r = 0; r < currentPiece.shape.length; r++) {
                    for (let c = 0; c < currentPiece.shape[r].length; c++) {
                        if (currentPiece.shape[r][c]) {
                            drawGhostBlock(
                                ctx,
                                (currentPiece.x + c) * blockWidth,
                                (ghostY + r) * blockHeight,
                                blockWidth,
                                blockHeight,
                                currentPiece.type
                            );
                        }
                    }
                }

                // Renderizar Peça Atual
                for (let r = 0; r < currentPiece.shape.length; r++) {
                    for (let c = 0; c < currentPiece.shape[r].length; c++) {
                        if (currentPiece.shape[r][c]) {
                            drawBlock(
                                ctx,
                                (currentPiece.x + c) * blockWidth,
                                (currentPiece.y + r) * blockHeight,
                                blockWidth,
                                blockHeight,
                                currentPiece.type
                            );
                        }
                    }
                }
            }

            // Renderizar partículas
            drawParticles(ctx);
        }

        // Helper para compatibilidade de bordas arredondadas em canvas
        function drawRoundRect(context, x, y, w, h, r) {
            if (context.roundRect) {
                context.roundRect(x, y, w, h, r);
            } else {
                context.rect(x, y, w, h);
            }
        }

        // Desenha um Bloco Lúdico 3D Estilizado com Gradientes e Bordas Arredondadas
        function drawBlock(context, x, y, w, h, type) {
            const colors = PALETTE[type] || { main: '#a855f7', light: '#d8b4fe', dark: '#7e22ce' };
            const radius = 4;

            context.save();

            // Sombra leve sob o bloco
            context.fillStyle = colors.dark;
            context.beginPath();
            drawRoundRect(context, x, y, w, h, radius);
            context.fill();

            // Corpo Principal com Gradiente
            const grad = context.createLinearGradient(x, y, x + w, y + h);
            grad.addColorStop(0, colors.light);
            grad.addColorStop(0.5, colors.main);
            grad.addColorStop(1, colors.dark);

            context.fillStyle = grad;
            context.beginPath();
            drawRoundRect(context, x + 1, y + 1, w - 2, h - 2, radius);
            context.fill();

            // Brilho no topo (Efeito 3D Cristalino)
            context.fillStyle = 'rgba(255, 255, 255, 0.35)';
            context.beginPath();
            drawRoundRect(context, x + 2, y + 2, w - 4, (h - 4) / 3, radius / 2);
            context.fill();

            context.restore();
        }

        // Desenha a Peça Sombra (Fantasma)
        function drawGhostBlock(context, x, y, w, h, type) {
            const colors = PALETTE[type] || { main: '#fff' };
            context.save();
            context.strokeStyle = colors.light;
            context.lineWidth = 1.5;
            context.fillStyle = 'rgba(255, 255, 255, 0.08)';
            
            context.beginPath();
            drawRoundRect(context, x + 1, y + 1, w - 2, h - 2, 4);
            context.fill();
            context.stroke();
            context.restore();
        }

        // Renderização dos Canvases Secundários (Próxima e Guardar)
        function drawPreviewCanvas(context, pieceType) {
            if (!context) return;
            context.clearRect(0, 0, context.canvas.width, context.canvas.height);
            if (!pieceType) return;

            const shape = typeof pieceType === 'string' ? SHAPES[pieceType] : pieceType.shape;
            const type = typeof pieceType === 'string' ? pieceType : pieceType.type;

            if (!shape || !shape.length || !shape[0]) return;

            const rows = shape.length;
            const cols = shape[0].length;

            const cellSize = Math.min(context.canvas.width / (cols + 1), context.canvas.height / (rows + 1));
            const offsetX = (context.canvas.width - cols * cellSize) / 2;
            const offsetY = (context.canvas.height - rows * cellSize) / 2;

            for (let r = 0; r < rows; r++) {
                for (let c = 0; c < cols; c++) {
                    if (shape[r][c]) {
                        drawBlock(
                            context,
                            offsetX + c * cellSize,
                            offsetY + r * cellSize,
                            cellSize,
                            cellSize,
                            type
                        );
                    }
                }
            }
        }

        /* ==========================================================================
           EFEITOS VISUAIS: PARTÍCULAS E TREMOR DE TELA
           ========================================================================== */
        function createParticles(x, y, color) {
            for (let i = 0; i < 8; i++) {
                particles.push({
                    x: x,
                    y: y,
                    vx: (Math.random() - 0.5) * 6,
                    vy: (Math.random() - 0.5) * 6 - 2,
                    color: color,
                    size: Math.random() * 4 + 2,
                    alpha: 1,
                    life: 1
                });
            }
        }

        function updateParticles() {
            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                p.x += p.vx;
                p.y += p.vy;
                p.vy += 0.15; // Gravidade
                p.alpha -= 0.025;
                p.life -= 0.025;

                if (p.alpha <= 0) {
                    particles.splice(i, 1);
                }
            }
        }

        function drawParticles(context) {
            context.save();
            particles.forEach(p => {
                context.globalAlpha = Math.max(0, p.alpha);
                context.fillStyle = p.color;
                context.beginPath();
                context.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                context.fill();
            });
            context.restore();
        }

        function triggerScreenShake() {
            const wrapper = document.getElementById('canvasWrapper');
            wrapper.classList.remove('shake');
            void wrapper.offsetWidth; // Trigger reflow
            wrapper.classList.add('shake');
        }

        function setupKeyboard() {
            document.addEventListener('keydown', (e) => {
                if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') {
                    playerMove(-1);
                } else if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') {
                    playerMove(1);
                } else if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') {
                    playerDrop();
                } else if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') {
                    playerRotate();
                } else if (e.code === 'Space') {
                    e.preventDefault();
                    playerHardDrop();
                } else if (e.key === 'c' || e.key === 'C' || e.key === 'Shift') {
                    playerHold();
                } else if (e.key === 'p' || e.key === 'P' || e.key === 'Escape') {
                    togglePause();
                }
            });
        }

        function setupTouchControls() {
            const bindBtn = (id, action) => {
                const btn = document.getElementById(id);
                if (!btn) return;

                let interval = null;

                const startAction = (e) => {
                    e.preventDefault();
                    action();
                };

                btn.addEventListener('touchstart', startAction, { passive: false });
                btn.addEventListener('mousedown', startAction);
            };

            bindBtn('btnLeft', () => playerMove(-1));
            bindBtn('btnRight', () => playerMove(1));
            bindBtn('btnRotate', () => playerRotate());
            bindBtn('btnDown', () => playerDrop());
            bindBtn('btnHardDrop', () => playerHardDrop());
            bindBtn('btnHold', () => playerHold());
        }
